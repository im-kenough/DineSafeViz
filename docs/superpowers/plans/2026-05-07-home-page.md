import sys
import os
from datetime import date, datetime, timedelta
from unittest.mock import patch, MagicMock

sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))
import app as app_module


def _reset_cache():
    app_module._stats_cache["data"] = None
    app_module._stats_cache["fetched_at"] = None


def _mock_stats_db(total, min_date, max_date):
    """Return a psycopg2 connection mock for _get_home_stats queries."""
    mock_conn = MagicMock()
    mock_cur = MagicMock()
    mock_conn.cursor.return_value = mock_cur
    mock_cur.fetchone.side_effect = [
        (total,),
        (min_date, max_date),
    ]
    return mock_conn


def test_get_home_stats_returns_correct_values():
    _reset_cache()
    mock_conn = _mock_stats_db(12345, date(2001, 1, 1), date(2025, 12, 31))
    with patch("app.psycopg2.connect", return_value=mock_conn):
        stats = app_module._get_home_stats()
    assert stats["total_inspections"] == 12345
    assert stats["years_of_data"] == 25  # 2025 - 2001 + 1


def test_get_home_stats_uses_cache_on_second_call():
    _reset_cache()
    mock_conn = _mock_stats_db(12345, date(2001, 1, 1), date(2025, 12, 31))
    with patch("app.psycopg2.connect", return_value=mock_conn):
        result1 = app_module._get_home_stats()
        result2 = app_module._get_home_stats()
    assert result1 == result2
    # DB connection opened exactly once
    assert mock_conn.cursor.call_count == 1


def test_get_home_stats_refreshes_after_ttl():
    # Seed an expired cache entry
    app_module._stats_cache["data"] = {"total_inspections": 999, "years_of_data": 10}
    app_module._stats_cache["fetched_at"] = datetime.now() - timedelta(days=6)

    mock_conn = _mock_stats_db(12345, date(2001, 1, 1), date(2025, 12, 31))
    with patch("app.psycopg2.connect", return_value=mock_conn):
        result = app_module._get_home_stats()

    assert result["total_inspections"] == 12345
    assert mock_conn.cursor.call_count == 1

```python
_stats_cache: dict = {"data": None, "fetched_at": None}
_STATS_TTL = timedelta(days=5)


def _get_home_stats() -> dict:
    """Return total inspection count and years of data, cached for 5 days."""
    now = datetime.now()
    if _stats_cache["data"] is None or (now - _stats_cache["fetched_at"]) > _STATS_TTL:
        conn = psycopg2.connect(**DB_CONFIG)
        cur = conn.cursor()
        cur.execute("SELECT COUNT(*) FROM inspections")
        total = cur.fetchone()[0]
        cur.execute("SELECT MIN(inspection_date), MAX(inspection_date) FROM inspections")
        min_date, max_date = cur.fetchone()
        cur.close()
        conn.close()
        years = (max_date.year - min_date.year + 1) if min_date and max_date else 0
        _stats_cache["data"] = {"total_inspections": total, "years_of_data": years}
        _stats_cache["fetched_at"] = now
    return _stats_cache["data"]
```
